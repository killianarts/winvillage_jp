# ChoiceField and IntegerField interactions

I had a hard day because I didn't properly understand how to use
choices with models and regular forms (rather than ModelForms).

Originally, I tried using the IntegerChoices enum

```python
from django.db import models
from django import forms


# models.py
class SomeModel(models.Model):
    class SomeChoices(models.IntegerChoices):
        ZERO = 0
        ONE = 1
        TWO = 2

    some_field = models.IntegerField(choices=SomeChoices)

    def set_some_field(self, form):
        some_field = form.cleaned_data["some_field"]
        self.some_field = self.SomeChoices[some_field]
        self.save()


# forms.py
class SomeForm(forms.Form):
    # choices duplicated here
    CHOICES = (("ZERO", 0), ("ONE", 1), ("TWO", 0))
    some_field = forms.ChoiceField(choices=CHOICES)


# views.py
def some_view(request):
    some_model_obj = get_or_set_session(request)  # Get obj by session id
    initial = {"some_field": my_obj.some_field}
    form = SomeForm(initial=initial)
    if request.method == "POST":
        form = SomeForm(request.POST)
        if form.is_valid():
            some_model_obj.set_some_field(form)
```

Initially, the problem manifested as the form rendering with the
default value even after `some_field` was set with `set_some_field` and
the page was reloaded.

```python
    initial = {"some_field": my_obj.some_field}
```

The problem is that the model `some_field` is an IntegerField returning
an integer value, but the form's `CHOICES` expects "ZERO", etc.

I tried resolving using `get_some_field_display()`.

```python
    initial = {"some_field": my_obj.get_some_field_display()}
```

That worked for the form, but now I got a key error when trying to set
a new value.

```python
    def set_some_field(self, form):


some_field = form.cleaned_data["some_field"]
self.some_field = self.SomeChoices[some_field]  # Problem
self.save()
```

The problem is `SomeChoices[some_field]` looked for `SomeChoices[1]`,
but it expected `SomeChoices["ONE"]`. I stupidly used SomeChoices
when it wasn't necessary.

```python
    def set_some_field(self, form):


self.some_field = form.cleaned_data["some_field"]
self.save()
```

This fixed the problem, but I didn't want the form to render "ZERO",
"ONE", "TWO". I wanted it to render "0", "1", "2".

```python
from django.db import models
from django import forms

CHOICES = ((0, "0"), (1, "1"), (2, "2"))


# models.py
class SomeModel(models.Model):
    some_field = models.IntegerField(choices=CHOICES)

    def set_some_field(self, form):
        self.some_field = form.cleaned_data["some_field"]
        self.save()


# forms.py
from .models import CHOICES


class SomeForm(forms.Form):
    some_field = forms.ChoiceField(choices=CHOICES)


# views.py
def some_view(request):
    some_model_obj = get_or_set_session(request)  # Get obj by session id
    initial = {"some_field": my_obj.some_field}
    form = SomeForm(initial=initial)
    if request.method == "POST":
        form = SomeForm(request.POST)
        if form.is_valid():
            some_model_obj.set_some_field(form)
```
